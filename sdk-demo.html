import { mountBalatro, createStrategyRunner } from './sdk.js';
const game = mountBalatro(document.querySelector('#game'), { seed: 'HELLO', startInMenu: false });
const runner = createStrategyRunner(game, async (state, {signal}) => {
  if (state.phase === 'blind_select') return {type:'select_blind'};
  return null; // 返回操作，或暂停
});
await runner.start();
runner.stop();
game.destroy();
